#!/usr/bin/env node /** * GHL MCP Server * * An MCP server providing full-coverage tools for the GoHighLevel (GHL) API: * - Contacts: get, search, create, update, upsert, delete, tags, notes * - Opportunities: get, search, create, update, status, delete, pipelines * - Conversations: get, search, create, messages, send, update * - Calendars: list, get, slots, appointments (CRUD) * * Authentication: Set GHL_API_KEY environment variable to your Private Integration Token. */ import { McpServer } from "@modelcontextprotocol/sdk/server/mcp.js"; import { StdioServerTransport } from "@modelcontextprotocol/sdk/server/stdio.js"; import { registerContactTools } from "./tools/contacts.js"; import { registerOpportunityTools } from "./tools/opportunities.js"; import { registerConversationTools } from "./tools/conversations.js"; import { registerCalendarTools } from "./tools/calendars.js"; // ── Create server ────────────────────────────────────────────────────────── const server = new McpServer({ name: "ghl-mcp-server", version: "1.0.0", }); // ── Register all tool domains ────────────────────────────────────────────── registerContactTools(server); registerOpportunityTools(server); registerConversationTools(server); registerCalendarTools(server); // ── Start server ─────────────────────────────────────────────────────────── async function main() { const apiKey = process.env.GHL_API_KEY; if (!apiKey) { console.error("ERROR: GHL_API_KEY environment variable is required.\n" + "Set it to your GoHighLevel Private Integration Token.\n" + "Get one at: Settings > Integrations > Private Integrations in your GHL sub-account."); process.exit(1); } const transport = new StdioServerTransport(); await server.connect(transport); console.error("GHL MCP server running via stdio"); } main().catch((error) => { console.error("Server error:", error); process.exit(1); }); //# sourceMappingURL=index.js.map
